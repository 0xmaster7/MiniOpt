cmake_minimum_required(VERSION 3.16)
project(MiniOpt VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(miniopt_core src/miniopt.cpp)
target_include_directories(miniopt_core PUBLIC include)
add_executable(miniopt src/main.cpp)
target_link_libraries(miniopt PRIVATE miniopt_core)
add_executable(unit_tests tests/unit.cpp)
target_link_libraries(unit_tests PRIVATE miniopt_core)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME unit COMMAND unit_tests)
add_test(NAME differential COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test.py $<TARGET_FILE:miniopt>)
add_test(NAME benchmark_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/benchmarks/benchmark.py $<TARGET_FILE:miniopt> --smoke)
add_test(NAME audit_regressions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/remediation.py $<TARGET_FILE:miniopt>)
